# Runtime dependencies for the Research Paper Search tool.
# Install with:  pip install -r requirements.txt
#
# PDF text is read with pdfium (fast, C++), falling back to pypdf (pure Python)
# for files pdfium cannot open. Without rapidfuzz, fuzzy scoring falls back to
# the stdlib difflib. Installing all three gives the best speed and accuracy.

pypdfium2>=4.30.0    # primary PDF text extraction (pdfium, Apache-2.0/BSD-3)
pypdf>=4.0.0         # extraction fallback
rapidfuzz>=3.9.0     # fast fuzzy match scoring

# Optional: pdfplumber>=0.11.0 is used as an extra fallback if installed.
